令 $S\subseteq f(P),T\subseteq f(Q)$，我们计算答案后缀和，也就是 
$$
h(S,T)=\sum_{S\subseteq P',T\subseteq Q'}\mathrm{val}_{P',Q'}
$$
这部分是好计数的，不过别忘了 $P\cap Q=\emptyset$ 这个条件，有
$$
h(S,T)=\prod_{S\subseteq i,T\nsubseteq i}(a_i+1)\prod_{T\subseteq i,S\nsubseteq i}(a_i+1)\prod_{T\cup S\subseteq i}(2a_i+1)
$$
为了方便计数，可以转化成
$$
h(S,T)=\prod_{S\subseteq i}(a_i+1)\prod_{T\subseteq i}(a_i+1)\prod_{T\cup S\subseteq i}\frac{2a_i+1}{(a_i+1)^2}
$$
设
$$
f(S)=\prod_{S\subseteq i}(a_i+1)
$$

$$
g(S)=\prod_{S\subseteq i}\frac{2a_i+1}{(a_i+1)^2}
$$

容斥系数不知道啊，我们设其为 $F(S,T)$
$$
\mathrm{ans}=\sum_{S,T}F(S,T)f(S)f(T)g(S\cup T)=\sum_P\mathrm{val}_{P,P}
$$
我不会组合意义或者打表，但是我会大力算容斥系数，考虑求单个 $\mathrm{val}_{P,P}$，是最基础的容斥
$$
\mathrm{val}_{P,P}=\sum_{S,T\supseteq P} (-1)^{|S|+|T|-2|P|}h(S,T)
$$
观察发现对于 $h(S,T)$，若 $|S|+|T|$ 为奇数，则对于所有 $\mathrm{val}_{P,P}$ 的计算，它的系数都为 $-1$，偶数则反之，因为 $2|P|$ 恒为偶数。

对于所有 $\mathrm{val}_{P,P}$ 的计算，显然 $h(S,T)$ 的出现次数为 $2^{|S\cap T|}$ 次，于是我们得到了容斥系数 $F(S,T)=2^{|S\cap T|}(-1)^{|S|+|T|}$。把这个带回原式看看有什么新发现
$$
\mathrm{ans}=\sum_{S,T}2^{|S\cap T|}(-1)^{|S|+|T|}f(S)f(T)g(S\cup T)
$$
注意到 $|S|+|T|-|S\cup T|=|S\cap T|$，所以我们有
$$
\mathrm{ans}=\sum_{S,T}(-2)^{|S|}(-2)^{|T|}2^{-|S\cup T|}f(S)f(T)g(S\cup T)
$$
记 $f'(S)=(-2)^{|S|}f(S),g'(S)=2^{-|S|}g(S)$，带回原式
$$
\mathrm{ans}=\sum_{S,T}f'(S)f'(T)g'(S\cup T)
$$
对 $f'$ 做 OR 卷积，和 $g'$ 对位相乘，发现做完了。复杂度 $O(2^nn)$。

然后发现只能过 B 性质，因为 $a_i+1\equiv 0$ 没逆元。考虑存数的时候计 $a\times0^b$，加法的时候留 $b$ 小的那个，乘法就 $b$ 相加。

using namespace std;
#define rep(i, x, y) for (int i = (x); i <= (y); i++)
#define per(i, x, y) for (int i = (x); i >= (y); i--)
#define int long long
inline int read() {
    char c = getchar(); int s = 0, f = 1;
    while (!isdigit(c)) { if (c == '-') f = -1; c = getchar(); }
    while (isdigit(c)) s = (s << 3) + (s << 1) + (c ^ 48), c = getchar();
    return s * f;
}
constexpr int N = 1 << 20, mod = 998244353, inv2 = 499122177;
int n, m, a[N], pw[25], pwi[25];
struct nint { int a, b; } f[N], g[N];
nint mul(const nint x, const nint y) {
    return {x.a * y.a % mod, x.b + y.b};
}
nint add(const nint x, const nint y) {
    if (x.b == y.b) return {(x.a + y.a) % mod, x.b};
    return x.b > y.b ? y : x;
}
int qpow(int x, int y = mod - 2) {
    int res = 1;
    while (y) {
        if (y & 1) (res *= x) %= mod;
        (x *= x) %= mod, y >>= 1;
    }
    return res;
}
void OR(nint *f, int o = 1) {
    for (int w = 1; w < n; w <<= 1)
        for (int i = 0; i < n; i += (w << 1)) rep (j, i, i + w - 1) 
            f[j + w] = add(f[j + w], {f[j].a * o % mod, f[j].b});
}
void sol() {
    n = 1 << (m = read());
    rep (i, 0, n - 1) {
        a[i] = read();
        int x = (a[i] + 1) % mod, y = (2 * a[i] + 1) % mod;
        if (x) {
            f[i] = {x, 0};
            if (y) g[i] = {y * qpow(x * x % mod) % mod, 0};
            else g[i] = {qpow(x * x % mod), 1};
        } else {
            f[i] = {1, 1};
            if (y) g[i] = {y, -2};
            else g[i] = {1, -1};
        }
    }
    rep (i, 0, m - 1) rep (S, 0, n - 1)
        if (!((S >> i) & 1)) {
            f[S] = mul(f[S], f[S | (1 << i)]);
            g[S] = mul(g[S], g[S | (1 << i)]);
        }
    rep (S, 0, n - 1) {
        (f[S].a *= pw[__builtin_popcount(S)]) %= mod; 
        (g[S].a *= pwi[__builtin_popcount(S)]) %= mod;
    }
    OR(f, 1);
    rep (i, 0, n - 1) f[i] = mul(f[i], f[i]);
    OR(f, mod - 1);
    int ans = 0;
    rep (S, 0, n - 1) {
        nint x = mul(f[S], g[S]);
        if (x.b == 0) (ans += x.a) %= mod;
    }
    cout << ans << '\n';
}
